-- =========================================================
-- BASE DE DATOS: capitulos_tecnm_morelia (version mejorada FIX)
-- MySQL 8+
-- =========================================================

DROP DATABASE IF EXISTS capitulos_tecnm_morelia;
CREATE DATABASE capitulos_tecnm_morelia
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_0900_ai_ci;

USE capitulos_tecnm_morelia;

-- =========================================================
-- TABLAS BASE: ROLES Y MIEMBROS
-- =========================================================

CREATE TABLE roles (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(50) NOT NULL UNIQUE,
  descripcion VARCHAR(255),
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

INSERT INTO roles (nombre, descripcion) VALUES
  ('Miembro', 'Miembro base del capitulo'),
  ('Asesor', 'Miembro asesor del capitulo'),
  ('Encargado', 'Miembro encargado con permisos de gestion del capitulo'),
  ('Admin', 'Administrador del capitulo');

CREATE TABLE miembros (
  id INT AUTO_INCREMENT PRIMARY KEY,
  numero_control VARCHAR(30),
  nombre VARCHAR(80) NOT NULL,
  apellidos VARCHAR(120) NOT NULL,
  username VARCHAR(80) NOT NULL UNIQUE,
  email VARCHAR(190) UNIQUE,
  telefono VARCHAR(30),
  password_hash VARCHAR(255) NOT NULL,
  foto_perfil MEDIUMTEXT,
  activo TINYINT(1) DEFAULT 1,
  verificado TINYINT(1) DEFAULT 0,
  tecnologico VARCHAR(150),

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  eliminado_en TIMESTAMP NULL DEFAULT NULL
) ENGINE=InnoDB;

CREATE INDEX idx_miembros_numero_control ON miembros(numero_control);
CREATE INDEX idx_miembros_activo ON miembros(activo);

CREATE TABLE miembros_roles (
  id INT AUTO_INCREMENT PRIMARY KEY,
  miembro_id INT NOT NULL,
  rol_id INT NOT NULL,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE (miembro_id, rol_id),
  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE CASCADE,
  FOREIGN KEY (rol_id) REFERENCES roles(id) ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE INDEX idx_miembros_roles_miembro ON miembros_roles(miembro_id);
CREATE INDEX idx_miembros_roles_rol ON miembros_roles(rol_id);

-- =========================================================
-- CAPITULOS
-- =========================================================

CREATE TABLE capitulos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  clave VARCHAR(30) UNIQUE,
  nombre VARCHAR(150) NOT NULL,
  descripcion TEXT,
  area VARCHAR(150),
  logo VARCHAR(255),
  color VARCHAR(30),
  activo TINYINT(1) DEFAULT 1,

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  eliminado_en TIMESTAMP NULL DEFAULT NULL,

  creado_por INT NULL,
  actualizado_por INT NULL,

  FOREIGN KEY (creado_por) REFERENCES miembros(id) ON DELETE SET NULL,
  FOREIGN KEY (actualizado_por) REFERENCES miembros(id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE INDEX idx_capitulos_activo ON capitulos(activo);
CREATE INDEX idx_capitulos_area ON capitulos(area);

-- =========================================================
-- MIEMBROS EN CAPITULOS
-- =========================================================

CREATE TABLE capitulos_miembros (
  id INT AUTO_INCREMENT PRIMARY KEY,
  capitulo_id INT NOT NULL,
  miembro_id INT NOT NULL,
  estado ENUM('Activo','Inactivo') DEFAULT 'Activo',
  motivo VARCHAR(255),

  fecha_ingreso TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE (capitulo_id, miembro_id),
  FOREIGN KEY (capitulo_id) REFERENCES capitulos(id) ON DELETE CASCADE,
  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE INDEX idx_capitulos_miembros_capitulo ON capitulos_miembros(capitulo_id);
CREATE INDEX idx_capitulos_miembros_miembro ON capitulos_miembros(miembro_id);

-- =========================================================
-- ASESORES / RESPONSABLES EN CAPITULOS
-- =========================================================

CREATE TABLE capitulos_asesores (
  id INT AUTO_INCREMENT PRIMARY KEY,
  capitulo_id INT NOT NULL,
  asesor_id INT NOT NULL,
  tipo ENUM('Asesor','Encargado') DEFAULT 'Asesor',

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE (capitulo_id, asesor_id),
  FOREIGN KEY (capitulo_id) REFERENCES capitulos(id) ON DELETE CASCADE,
  FOREIGN KEY (asesor_id) REFERENCES miembros(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE INDEX idx_capitulos_asesores_capitulo ON capitulos_asesores(capitulo_id);
CREATE INDEX idx_capitulos_asesores_asesor ON capitulos_asesores(asesor_id);

-- =========================================================
-- ACTIVIDADES
-- =========================================================

CREATE TABLE actividades (
  id INT AUTO_INCREMENT PRIMARY KEY,
  capitulo_id INT NOT NULL,
  tipo ENUM('Curso','Taller','Evento','Conferencia') NOT NULL,

  titulo VARCHAR(200) NOT NULL,
  descripcion TEXT,
  lugar VARCHAR(200),
  cupo INT,

  otorga_creditos TINYINT(1) DEFAULT 0,
  creditos INT DEFAULT 0,

  fecha_inicio DATETIME NOT NULL,
  fecha_fin DATETIME NOT NULL,

  estado ENUM('Borrador','Publicado','Cancelado','Finalizado') DEFAULT 'Borrador',

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  eliminado_en TIMESTAMP NULL DEFAULT NULL,

  creado_por INT NULL,
  actualizado_por INT NULL,

  FOREIGN KEY (capitulo_id) REFERENCES capitulos(id) ON DELETE CASCADE,
  FOREIGN KEY (creado_por) REFERENCES miembros(id) ON DELETE SET NULL,
  FOREIGN KEY (actualizado_por) REFERENCES miembros(id) ON DELETE SET NULL,

  CHECK (fecha_fin >= fecha_inicio),
  CHECK (creditos >= 0),
  CHECK (
    (otorga_creditos = 0 AND creditos = 0)
    OR
    (otorga_creditos = 1 AND creditos > 0)
  )
) ENGINE=InnoDB;

CREATE INDEX idx_actividades_capitulo ON actividades(capitulo_id);
CREATE INDEX idx_actividades_fechas ON actividades(fecha_inicio, fecha_fin);
CREATE INDEX idx_actividades_estado ON actividades(estado);
CREATE INDEX idx_actividades_otorga_creditos ON actividades(otorga_creditos);

-- =========================================================
-- SESIONES DE UNA ACTIVIDAD
-- =========================================================

CREATE TABLE actividades_sesiones (
  id INT AUTO_INCREMENT PRIMARY KEY,
  actividad_id INT NOT NULL,

  titulo VARCHAR(200),
  descripcion VARCHAR(255),

  fecha_inicio DATETIME NOT NULL,
  fecha_fin DATETIME NOT NULL,

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  CHECK (fecha_fin >= fecha_inicio)
) ENGINE=InnoDB;

CREATE INDEX idx_actividades_sesiones_actividad ON actividades_sesiones(actividad_id);
CREATE INDEX idx_actividades_sesiones_fechas ON actividades_sesiones(fecha_inicio, fecha_fin);

-- =========================================================
-- RESPONSABLES DE ACTIVIDADES (INTERNOS O EXTERNOS)
-- =========================================================

CREATE TABLE actividades_responsables (
  id INT AUTO_INCREMENT PRIMARY KEY,
  actividad_id INT NOT NULL,
  nombre VARCHAR(150) NOT NULL,
  email VARCHAR(190),

  es_externo TINYINT(1) DEFAULT 0,
  miembro_id INT NULL,

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE INDEX idx_actividades_responsables_actividad ON actividades_responsables(actividad_id);
CREATE INDEX idx_actividades_responsables_miembro ON actividades_responsables(miembro_id);

-- =========================================================
-- INSCRIPCIONES A ACTIVIDADES
-- =========================================================

CREATE TABLE actividades_inscripciones (
  id INT AUTO_INCREMENT PRIMARY KEY,
  actividad_id INT NOT NULL,
  miembro_id INT NOT NULL,

  estado ENUM('Inscrito','Cancelado','Finalizado') DEFAULT 'Inscrito',

  fecha_inscripcion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE (actividad_id, miembro_id),
  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE INDEX idx_actividades_inscripciones_actividad ON actividades_inscripciones(actividad_id);
CREATE INDEX idx_actividades_inscripciones_miembro ON actividades_inscripciones(miembro_id);
CREATE INDEX idx_actividades_inscripciones_estado ON actividades_inscripciones(estado);

-- =========================================================
-- ASISTENCIA POR SESION
-- =========================================================

CREATE TABLE actividades_asistencia_sesiones (
  id INT AUTO_INCREMENT PRIMARY KEY,

  actividad_id INT NOT NULL,
  sesion_id INT NOT NULL,
  miembro_id INT NOT NULL,

  asistio TINYINT(1) DEFAULT 0,
  registrado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  UNIQUE (sesion_id, miembro_id),

  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  FOREIGN KEY (sesion_id) REFERENCES actividades_sesiones(id) ON DELETE CASCADE,
  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE INDEX idx_asistencia_sesion_sesion ON actividades_asistencia_sesiones(sesion_id);
CREATE INDEX idx_asistencia_sesion_miembro ON actividades_asistencia_sesiones(miembro_id);
CREATE INDEX idx_asistencia_sesion_actividad ON actividades_asistencia_sesiones(actividad_id);

-- =========================================================
-- EVIDENCIAS DE ACTIVIDADES
-- =========================================================

CREATE TABLE actividades_evidencias (
  id INT AUTO_INCREMENT PRIMARY KEY,
  actividad_id INT NOT NULL,

  archivo VARCHAR(255) NOT NULL,
  descripcion VARCHAR(255),
  subido_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

  subido_por INT NULL,

  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  FOREIGN KEY (subido_por) REFERENCES miembros(id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE INDEX idx_actividades_evidencias_actividad ON actividades_evidencias(actividad_id);

-- =========================================================
-- AVISOS
-- =========================================================

CREATE TABLE avisos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  capitulo_id INT NULL,

  titulo VARCHAR(200) NOT NULL,
  contenido TEXT NOT NULL,
  prioridad ENUM('Baja','Normal','Alta') DEFAULT 'Normal',

  publicado TINYINT(1) DEFAULT 0,
  fecha_publicacion DATETIME NULL,

  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  eliminado_en TIMESTAMP NULL DEFAULT NULL,

  creado_por INT NULL,
  actualizado_por INT NULL,

  FOREIGN KEY (capitulo_id) REFERENCES capitulos(id) ON DELETE CASCADE,
  FOREIGN KEY (creado_por) REFERENCES miembros(id) ON DELETE SET NULL,
  FOREIGN KEY (actualizado_por) REFERENCES miembros(id) ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE INDEX idx_avisos_capitulo ON avisos(capitulo_id);
CREATE INDEX idx_avisos_publicado ON avisos(publicado);
CREATE INDEX idx_avisos_fecha ON avisos(fecha_publicacion);
CREATE INDEX idx_avisos_prioridad ON avisos(prioridad);

-- =========================================================
-- HISTORIAL DE CREDITOS POR MIEMBRO
-- =========================================================

CREATE TABLE miembros_creditos_historial (
  id INT AUTO_INCREMENT PRIMARY KEY,

  miembro_id INT NOT NULL,
  actividad_id INT NOT NULL,

  creditos_otorgados INT NOT NULL,
  motivo VARCHAR(255),

  otorgado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  otorgado_por INT NULL,

  UNIQUE (miembro_id, actividad_id),

  FOREIGN KEY (miembro_id) REFERENCES miembros(id) ON DELETE CASCADE,
  FOREIGN KEY (actividad_id) REFERENCES actividades(id) ON DELETE CASCADE,
  FOREIGN KEY (otorgado_por) REFERENCES miembros(id) ON DELETE SET NULL,

  CHECK (creditos_otorgados > 0)
) ENGINE=InnoDB;

CREATE INDEX idx_creditos_historial_miembro ON miembros_creditos_historial(miembro_id);
CREATE INDEX idx_creditos_historial_actividad ON miembros_creditos_historial(actividad_id);
CREATE INDEX idx_creditos_historial_fecha ON miembros_creditos_historial(otorgado_en);

DELIMITER ;
